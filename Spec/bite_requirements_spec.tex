\section{Introduction}

\subsection{Purpose}
This document defines the requirements for \textit{Bite}, a multi-page static website built
with HTML5 and a single external CSS file. It sets out what the site must do, how it must
look and the criteria used to accept it. It also maps every requirement to the SE 310
brief and grading rubric, so nothing in the brief is missed.

\subsection{Scope}
\textit{Bite} presents a photographic aesthetic built on raw, unpolished imagery: hard flash,
film grain, fisheye lenses, desaturated film tones and subjects that feel feral or calm in
the middle of chaos. Every image is paired with one short interpretive sentence.

\textbf{In scope:} four linked HTML pages, one stylesheet, local media (images and audio),
a contact form, W3C validation, and cross-browser testing.

\textbf{Out of scope:} JavaScript, frameworks, back-end form processing, a CMS and hosting.

\subsection{Definitions}
\begin{description}[leftmargin=2.5cm, style=nextline]
  \item[Interpretation] A single short sentence (12 words or fewer) that reads an image.
  \item[MUST] The requirement is mandatory for acceptance.
  \item[SHOULD] The requirement is strongly recommended but not mandatory.
  \item[COULD] The requirement is a stretch goal.
\end{description}

% =========================================================
\section{Concept}

\subsection{Creative Direction}
The site works like a printed photo zine: a dark canvas, heavy type and lots of empty
space. The writing is short and declarative, and it never explains too much. The tone is
confident, slightly menacing, and quiet.

\subsection{Sample Interpretations}
\begin{itemize}
  \item Snarling dog under flash --- \textit{``Not angry. Honest.''}
  \item Figure on fire carrying a briefcase --- \textit{``She's still going to the meeting.''}
  \item Empty road, parked motorcycle --- \textit{``The road doesn't ask where you're going.''}
  \item Pyramid with a lone figure --- \textit{``Some things were here before the question.''}
  \item Fisheye portrait --- \textit{``Distortion is just closeness.''}
\end{itemize}

\subsection{Target Audience}
Visitors interested in photography, visual culture and design, mainly on mobile and
desktop browsers.

\subsection{Layout \& Spacing}
\begin{itemize}
  \item A spacing scale based on a \texttt{0.5rem} unit (\texttt{-{}-space-1} to \texttt{-{}-space-6}).
  \item Maximum content width of \texttt{1200px}, centred.
  \item The gallery uses CSS \texttt{columns} for a masonry effect: 3 columns on desktop,
        2 on tablet and 1 on mobile.
  \item A grain texture overlay applied with a \texttt{body::before} pseudo-element.
\end{itemize}

\subsection{Imagery Treatment}
Images appear at reduced saturation (\texttt{filter: saturate(0.6)}) and return to full
colour on hover or focus, with a short CSS transition.

\section{Functional Requirements}

\subsection{Structure \& Text (Rubric: HTML Structure \& Text Formatting)}
\begin{longtable}{@{}p{1.6cm}p{1.6cm}p{12.3cm}@{}}
\toprule
\textbf{ID} & \textbf{Priority} & \textbf{Requirement} \\
\midrule
\endhead
\req{FR-01}{MUST}{Every page uses \texttt{<header>}, \texttt{<nav>}, \texttt{<main>}, \texttt{<section>} and \texttt{<footer>}.}
\req{FR-02}{MUST}{Each page has exactly one \texttt{<h1>}. \texttt{<h2>} and \texttt{<h3>} follow a logical hierarchy.}
\req{FR-03}{MUST}{The site uses bold, italic, underline, varied colours and varied text sizes, applied with semantic tags (\texttt{<strong>}, \texttt{<em>}, \texttt{<u>}) and CSS.}
\req{FR-04}{MUST}{Code uses consistent 2-space indentation and includes comments marking each major section.}
\req{FR-05}{MUST}{Every page declares \texttt{<!DOCTYPE html>}, \texttt{lang="en"}, a UTF-8 charset and a unique \texttt{<title>}.}
\bottomrule
\end{longtable}

\subsection{Elements, Media \& Form (Rubric: HTML Elements, Media \& Form)}
\begin{longtable}{@{}p{1.6cm}p{1.6cm}p{12.3cm}@{}}
\toprule
\textbf{ID} & \textbf{Priority} & \textbf{Requirement} \\
\midrule
\endhead
\req{FR-06}{MUST}{The site includes at least one \texttt{<ul>} (e.g. the navigation) and one \texttt{<ol>} (Home: ``How to look at it'').}
\req{FR-07}{MUST}{Internal links connect all four pages through the navigation bar.}
\req{FR-08}{MUST}{At least one external link uses \texttt{target="\_blank"} and \texttt{rel="noopener noreferrer"}.}
\req{FR-09}{MUST}{Every \texttt{<img>} has descriptive \texttt{alt} text. Gallery images sit in \texttt{<figure>} with the interpretation in \texttt{<figcaption>}.}
\req{FR-10}{MUST}{The Look page has a table with a \texttt{<caption>}, a \texttt{<thead>} with \texttt{<th scope="col">}, and at least 4 rows. Columns: Technique, Effect, Feeling.}
\req{FR-11}{MUST}{A local audio file plays through \texttt{<audio controls>}, with fallback text. Embedded YouTube does not satisfy this requirement.}
\req{FR-12}{MUST}{The contact form has labelled fields for name, email and message, plus a submit button.}
\req{FR-13}{MUST}{Form fields use HTML5 validation: \texttt{required}, \texttt{type="email"} and \texttt{minlength} on the message.}
\req{FR-14}{MUST}{Every label is linked to its input via \texttt{for}/\texttt{id}.}
\req{FR-15}{SHOULD}{The form uses a \texttt{mailto:} action or a documented placeholder action, so submission is testable.}
\req{FR-16}{MUST}{No empty pages, dead links or placeholder \texttt{href="\#"} values.}
\bottomrule
\end{longtable}

\subsection{Visual Design (Rubric: CSS Design \& Layout)}
\begin{longtable}{@{}p{1.6cm}p{1.6cm}p{12.3cm}@{}}

\textbf{ID} & \textbf{Priority} & \textbf{Requirement} \\

\section{Non-Functional Requirements}

\begin{longtable}{@{}p{1.6cm}p{1.6cm}p{12.3cm}@{}}
\toprule
\textbf{ID} & \textbf{Priority} & \textbf{Requirement} \\
\midrule
\endhead
\multicolumn{3}{@{}l}{\textit{Responsiveness \& Compatibility (Rubric: Cross-Browser Functionality)}} \\
\req{NFR-01}{MUST}{Every page includes \texttt{<meta name="viewport" content="width=device-width, initial-scale=1">}.}
\req{NFR-02}{MUST}{Layout adapts at breakpoints of 600px and 900px, with no horizontal scrolling at 360px width.}
\req{NFR-03}{MUST}{The site works on at least two browsers across two operating systems (e.g. Chrome on Windows 11 and Safari on iOS).}
\req{NFR-04}{SHOULD}{The site is also tested on Firefox and Edge.}
\midrule
\multicolumn{3}{@{}l}{\textit{Validation (Rubric: Validation \& Compatibility Report)}} \\
\req{NFR-05}{MUST}{Every HTML file passes the W3C Markup Validator with 0 errors.}
\req{NFR-06}{MUST}{\texttt{style.css} passes the W3C CSS Validator with 0 errors.}
\req{NFR-07}{SHOULD}{No warnings remain. Any that do are justified in the report.}
\midrule
\multicolumn{3}{@{}l}{\textit{Accessibility}} \\
\req{NFR-08}{MUST}{Text contrast meets WCAG AA (4.5:1 for body text) against \texttt{-{}-ink}.}
\req{NFR-09}{MUST}{All interactive elements are reachable by keyboard and show visible focus.}
\req{NFR-10}{SHOULD}{Animations and transitions respect \texttt{prefers-reduced-motion}.}
\req{NFR-11}{COULD}{The Lighthouse Accessibility score is 95 or higher on every page.}
\midrule
\multicolumn{3}{@{}l}{\textit{Performance \& Content}} \\
\req{NFR-12}{SHOULD}{Images are compressed (WebP or JPEG, 300KB or less each) and use \texttt{loading="lazy"} below the fold.}
\req{NFR-13}{MUST}{All images and audio are original or royalty-free (Unsplash, Pexels, Pixabay), with credits in the footer.}
\req{NFR-14}{MUST}{The site uses no JavaScript and no CSS or JS frameworks.}
\bottomrule
\end{longtable}

\section{Constraints \& Assumptions}
\begin{itemize}
  \item HTML5 and CSS3 only, with one external stylesheet.
  \item Web fonts may load from Google Fonts. If they fail, the site falls back gracefully to system fonts.
  \item The site runs from local files, so no server is required.
  \item Deadline: 24/11/2026. Submission is a \texttt{.zip} of the site folder plus a PDF report.
\end{itemize}

% =========================================================
\section{Acceptance Criteria}
The site is accepted when:
\begin{enumerate}
  \item All \textbf{MUST} requirements are met and verified.
  \item W3C validator screenshots show 0 errors for every HTML file and the CSS file.
  \item Compatibility testing across at least two browsers on two operating systems shows
        no unresolved \textit{Critical} or \textit{Major} issues.
  \item Every item on the brief's submission checklist is ticked.
\end{enumerate}
